Ejercicio 01 Nivel Básico

Cambiar color de fondo

Alterna dinámicamente el fondo de un contenedor mediante eventos de clic.

Color actual: #18181b
Ver código fuente 
<!-- HTML -->
<div id="contenedor">
  <p>Color: <span id="codigoColor">#18181b</span></p>
  <button id="btnCambiar">Cambiar color</button>
</div>
/* CSS */
#contenedor {
  padding: 2rem;
  border-radius: 8px;
  transition: background-color 0.3s ease;
}
#btnCambiar {
  padding: 10px 18px;
  font-weight: 500;
  cursor: pointer;
}
// JavaScript
const btnCambiar = document.getElementById('btnCambiar');
const contenedor = document.getElementById('contenedor');
const codigoColor = document.getElementById('codigoColor');

const colores = ['#27272a', '#1e293b', '#14532d', '#1e3a8a', '#581c87', '#701a75'];

btnCambiar.addEventListener('click', function() {
  const indiceAleatorio = Math.floor(Math.random() * colores.length);
  const nuevoColor = colores[indiceAleatorio];

  contenedor.style.backgroundColor = nuevoColor;
  codigoColor.textContent = nuevoColor;
});
Ejercicio 02 Nivel Básico

Lista dinámica en memoria

Almacena datos en un arreglo de JavaScript y renderiza la lista en el DOM.

Elementos en arreglo: 0
  • Sin elementos en memoria
Ver código fuente 
<!-- HTML -->
<input type="text" id="inputTexto" placeholder="Ingresa texto">
<button id="btnAgregar">Agregar</button>
<ul id="miLista"></ul>
/* CSS */
.dynamic-list { list-style: none; padding: 0; }
.dynamic-list li {
  background: #18181b;
  border: 1px solid #27272a;
  padding: 8px 12px;
  border-radius: 6px;
}
// JavaScript (Arreglos + DOM)
const datosEnMemoria = [];

function renderizarLista() {
  const lista = document.getElementById('miLista');
  lista.innerHTML = '';

  datosEnMemoria.forEach((item, index) => {
    const li = document.createElement('li');
    li.textContent = item;
    
    const btnEliminar = document.createElement('button');
    btnEliminar.textContent = 'Eliminar';
    btnEliminar.onclick = () => {
      datosEnMemoria.splice(index, 1);
      renderizarLista();
    };
    
    li.appendChild(btnEliminar);
    lista.appendChild(li);
  });
}
Ejercicio 03 Nivel Pro

Contador con límite

Incrementa valores y dispara un aviso del sistema al alcanzar 10.

Límite alcanzado (10)

Se ha cumplido la condición del contador.

0
Ver código fuente 
<!-- HTML -->
<button id="btnRestar">-</button>
<span id="valorContador">0</span>
<button id="btnSumar">+</button>
<div id="mensajeAlerta" style="display:none;">Límite alcanzado</div>
/* CSS */
.counter-number {
  font-size: 2.5rem;
  font-weight: 600;
}
.alert-box {
  border-left: 3px solid #f59e0b;
}
// JavaScript (Eventos y Condicionales)
let contador = 0;

function actualizarUI() {
  document.getElementById('valorContador').textContent = contador;
  
  if (contador === 10) {
    alert('Atención: El contador ha alcanzado el límite de 10.');
  }
}

document.getElementById('btnSumar').onclick = () => { contador++; actualizarUI(); };
document.getElementById('btnRestar').onclick = () => { contador--; actualizarUI(); };

.calc-grid {
  display: grid;
  grid-template-columns: repeat(4, 1fr);
  gap: 6px;
}
// Función de evaluación manual (Sin eval)
function calcular(a, operador, b) {
  const numA = parseFloat(a);
  const numB = parseFloat(b);
  if (isNaN(numA) || isNaN(numB)) return b;

  switch (operador) {
    case '+': return numA + numB;
    case '-': return numA - numB;
    case '*': case '×': return numA * numB;
    case '/': case '÷': return numB !== 0 ? numA / numB : 'Error';
    case '%': return numA % numB;
    default: return numB;
  }
}
